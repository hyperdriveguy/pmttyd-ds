cmake_minimum_required(VERSION 3.16)
project(ttyd_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(ttyd_core source/game.cpp source/save.cpp source/visibility.cpp source/uv_animation.cpp source/water_recovery.cpp source/battle_timing_a.cpp source/battle_hammer.cpp source/battle_defense.cpp source/battle_damage.cpp source/battle_status.cpp)
target_include_directories(ttyd_core PUBLIC include)
target_compile_options(ttyd_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE ttyd_core)
enable_testing()
add_test(NAME core COMMAND core_tests)
add_executable(battle_timing_a_tests tests/battle_timing_a_tests.cpp)
target_link_libraries(battle_timing_a_tests PRIVATE ttyd_core)
add_test(NAME battle_timing_a COMMAND battle_timing_a_tests)
add_executable(battle_hammer_tests tests/battle_hammer_tests.cpp)
target_link_libraries(battle_hammer_tests PRIVATE ttyd_core)
add_test(NAME battle_hammer COMMAND battle_hammer_tests)
add_executable(battle_defense_tests tests/battle_defense_tests.cpp)
target_link_libraries(battle_defense_tests PRIVATE ttyd_core)
add_test(NAME battle_defense COMMAND battle_defense_tests)
add_executable(battle_damage_tests tests/battle_damage_tests.cpp)
target_link_libraries(battle_damage_tests PRIVATE ttyd_core)
add_test(NAME battle_damage COMMAND battle_damage_tests)
add_executable(battle_status_tests tests/battle_status_tests.cpp)
target_link_libraries(battle_status_tests PRIVATE ttyd_core)
add_test(NAME battle_status COMMAND battle_status_tests)
